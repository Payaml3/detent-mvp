title: "Signature verification"
description: "Check that a delivery came from DekernAI and is recent, before you act on it."
---

Every delivery carries a signature header:

```http
DekernAI-Signature: t=1764756000,v1=5257a869e7ecebeda32affa62cdca3fa51cad7e77a0e56ff536d0ce8e108d8bd
```

`v1` is the hex HMAC-SHA256, with your endpoint's secret, of the timestamp, a full stop and the **raw** request
body: `"{t}.{raw_body}"`. To verify a delivery:

1. Read the raw body before any JSON parsing; a parsed and re-serialised body won't match.
2. Split the header on commas and take `t` and every `v1` (there can be more than one).
3. Compute the HMAC of `t + "." + raw_body` and compare it with each `v1` in constant time.
4. Reject the delivery if no signature matches, or if `t` is more than 300 seconds from your clock.
5. Answer `2xx` quickly, then process. Deduplicate on the event `id`.

Every sample below is a file in the docs repository that CI runs against the same test vectors (valid,
rotated, stale, future, tampered and malformed headers) before the docs are published. Each one uses only its
language's standard library.

```bash cURL
# verify_webhook SECRET HEADER BODY_FILE [NOW] [TOLERANCE]: exit 0 if the DekernAI-Signature header
# matches the raw body (read from a file, byte for byte) and is within the time window.
verify_webhook() {
  local secret=$1 header=$2 body_file=$3 now=${4:-$(date +%s)} tolerance=${5:-300}
  local timestamp="" part key value expected
  local -a signatures=()
  IFS=',' read -r -a parts <<< "$header"
  [[ $header == *, ]] && parts+=("")
  for part in "${parts[@]}"; do
    [[ $part == *=* ]] || return 1
    key=${part%%=*} value=${part#*=}
    key=${key//[[:space:]]/} value=${value//[[:space:]]/}
    if [[ $key == t ]]; then
      [[ -z $timestamp && $value =~ ^[0-9]+$ ]] || return 1
      timestamp=$value
    elif [[ $key == v1 ]]; then
      signatures+=("$value")
    fi
  done
  [[ -n $timestamp && ${#signatures[@]} -gt 0 ]] || return 1
  local age=$(( now - timestamp ))
  (( ${age#-} <= tolerance )) || return 1
  expected=$( { printf '%s.' "$timestamp"; cat "$body_file"; } \
    | openssl dgst -sha256 -hmac "$secret" -r | cut -d' ' -f1)
  for s in "${signatures[@]}"; do
    [[ $s == "$expected" ]] && return 0
  done
  return 1
}
```

```python Python
import hashlib
import hmac
import time


def verify_webhook(
    secret: str, header: str, body: bytes, now: int | None = None, tolerance: int = 300
) -> bool:
    """True if the DekernAI-Signature header matches the raw body and is within the time window."""
    timestamp, signatures = None, []
    for part in header.split(","):
        key, sep, value = part.strip().partition("=")
        if not sep:
            return False
        if key == "t":
            if timestamp is not None or not value.isdigit():
                return False
            timestamp = int(value)
        elif key == "v1":
            signatures.append(value)
    if timestamp is None or not signatures:
        return False
    if abs((int(time.time()) if now is None else now) - timestamp) > tolerance:
        return False
    expected = hmac.new(secret.encode(), f"{timestamp}.".encode() + body, hashlib.sha256).hexdigest()
    return any(hmac.compare_digest(expected, s) for s in signatures)
```

```typescript TypeScript
import { createHmac, timingSafeEqual } from "node:crypto";

export function verifyWebhook(
  secret: string,
  header: string,
  body: string | Buffer,
  now: number = Math.floor(Date.now() / 1000),
  tolerance = 300,
): boolean {
  let timestamp: number | undefined;
  const signatures: string[] = [];
  for (const part of header.split(",")) {
    const i = part.indexOf("=");
    if (i < 0) return false;
    const key = part.slice(0, i).trim();
    const value = part.slice(i + 1).trim();
    if (key === "t") {
      if (timestamp !== undefined || !/^\d+$/.test(value)) return false;
      timestamp = Number(value);
    } else if (key === "v1") {
      signatures.push(value);
    }
  }
  if (timestamp === undefined || signatures.length === 0) return false;
  if (Math.abs(now - timestamp) > tolerance) return false;
  const expected = Buffer.from(
    createHmac("sha256", secret).update(`${timestamp}.`).update(body).digest("hex"),
  );
  return signatures.some((s) => {
    const candidate = Buffer.from(s);
    return candidate.length === expected.length && timingSafeEqual(candidate, expected);
  });
}
```

```go Go
import (
	"crypto/hmac"
	"crypto/sha256"
	"encoding/hex"
	"strconv"
	"strings"
)

// VerifyWebhook reports whether the DekernAI-Signature header matches the raw body and is
// within the time window. Pass now as time.Now().Unix() and tolerance as 300.
func VerifyWebhook(secret, header string, body []byte, now, tolerance int64) bool {
	var timestamp int64 = -1
	var signatures []string
	for _, part := range strings.Split(header, ",") {
		key, value, found := strings.Cut(strings.TrimSpace(part), "=")
		if !found {
			return false
		}
		switch key {
		case "t":
			if timestamp >= 0 || value == "" || strings.Trim(value, "0123456789") != "" {
				return false
			}
			n, err := strconv.ParseInt(value, 10, 64)
			if err != nil {
				return false
			}
			timestamp = n
		case "v1":
			signatures = append(signatures, value)
		}
	}
	if timestamp < 0 || len(signatures) == 0 {
		return false
	}
	if now-timestamp > tolerance || timestamp-now > tolerance {
		return false
	}
	mac := hmac.New(sha256.New, []byte(secret))
	mac.Write([]byte(strconv.FormatInt(timestamp, 10) + "."))
	mac.Write(body)
	expected := []byte(hex.EncodeToString(mac.Sum(nil)))
	for _, s := range signatures {
		if hmac.Equal(expected, []byte(s)) {
			return true
		}
	}
	return false
}
```

```java Java
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.util.ArrayList;
import java.util.HexFormat;
import java.util.List;
import javax.crypto.Mac;
import javax.crypto.spec.SecretKeySpec;

/** True if the DekernAI-Signature header matches the raw body and is within the time window. */
public static boolean verifyWebhook(String secret, String header, byte[] body, long now, long tolerance)
        throws Exception {
    Long timestamp = null;
    List<String> signatures = new ArrayList<>();
    for (String part : header.split(",", -1)) {
        int i = part.indexOf('=');
        if (i < 0) return false;
        String key = part.substring(0, i).trim();
        String value = part.substring(i + 1).trim();
        if (key.equals("t")) {
            if (timestamp != null || !value.matches("[0-9]+")) return false;
            timestamp = Long.parseLong(value);
        } else if (key.equals("v1")) {
            signatures.add(value);
        }
    }
    if (timestamp == null || signatures.isEmpty()) return false;
    if (Math.abs(now - timestamp) > tolerance) return false;
    Mac mac = Mac.getInstance("HmacSHA256");
    mac.init(new SecretKeySpec(secret.getBytes(StandardCharsets.UTF_8), "HmacSHA256"));
    mac.update((timestamp + ".").getBytes(StandardCharsets.UTF_8));
    byte[] expected = HexFormat.of().formatHex(mac.doFinal(body)).getBytes(StandardCharsets.UTF_8);
    for (String s : signatures) {
        if (MessageDigest.isEqual(expected, s.getBytes(StandardCharsets.UTF_8))) return true;
    }
    return false;
}
```

```php PHP
/** True if the DekernAI-Signature header matches the raw body and is within the time window. */
function verify_webhook(string $secret, string $header, string $body, ?int $now = null, int $tolerance = 300): bool
{
    $timestamp = null;
    $signatures = [];
    foreach (explode(',', $header) as $part) {
        if (!str_contains($part, '=')) {
            return false;
        }
        [$key, $value] = array_map('trim', explode('=', $part, 2));
        if ($key === 't') {
            if ($timestamp !== null || !ctype_digit($value)) {
                return false;
            }
            $timestamp = (int) $value;
        } elseif ($key === 'v1') {
            $signatures[] = $value;
        }
    }
    if ($timestamp === null || !$signatures) {
        return false;
    }
    if (abs(($now ?? time()) - $timestamp) > $tolerance) {
        return false;
    }
    $expected = hash_hmac('sha256', $timestamp . '.' . $body, $secret);
    foreach ($signatures as $signature) {
        if (hash_equals($expected, $signature)) {
            return true;
        }
    }
    return false;
}
```

```ruby Ruby
require "openssl"

# True if the DekernAI-Signature header matches the raw body and is within the time window.
def verify_webhook(secret, header, body, now: Time.now.to_i, tolerance: 300)
  timestamp = nil
  signatures = []
  header.split(",", -1).each do |part|
    return false unless part.include?("=")

    key, value = part.split("=", 2).map(&:strip)
    if key == "t"
      return false if timestamp || !value.match?(/\A[0-9]+\z/)

      timestamp = value.to_i
    elsif key == "v1"
      signatures << value
    end
  end
  return false if timestamp.nil? || signatures.empty?
  return false if (now - timestamp).abs > tolerance

  expected = OpenSSL::HMAC.hexdigest("SHA256", secret, "#{timestamp}.#{body}")
  signatures.any? { |s| OpenSSL.secure_compare(expected, s) }
end
```

The samples take the current time as `now` and a 300-second `tolerance`. Pass the raw body exactly as received:
in most web frameworks that means reading the request bytes before any JSON middleware runs.
